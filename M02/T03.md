# Desenvolvimento de Aplicações Web Dinâmicas e Interativas

## Tópico 3: Context API

- O Problema: Prop Drilling
- O Conceito de Contexto: Provider e Consumer
- Receita dos 4 Passos
  - 1. Definir as Props Compartilhadas
  - 2. Criar o Contexto
  - 3. Criar o Provedor
  - 4. Criar o Hook de Consumo

### Estrutura do Contexto (Sugestão)

```tsx
import {
  type PropsWithChildren,
  createContext,
  useContext,
  useState,
} from "react";

// 1. Define your context props
export interface AppContextProps {
  track: number;
  prev: () => void;
  next: () => void;
  musicTime: number;
}

// 2. Create a new empty (undefined) context (you don't need to export it if using only the custom hook)
export const AppContext = createContext<AppContextProps | undefined>(undefined);

// 3. Create a custom context provider, so all context data will be self-contained
export default function AppContextProvider({ children }: PropsWithChildren) {
  const [track, setTrack] = useState(1);

  const appContext: AppContextProps = {
    track,
    prev: () => {
      if (track > 1) setTrack(track - 1);
    },
    next: () => {
      setTrack(track + 1);
    },
    musicTime: 1.55,
  };

  return (
    <AppContext.Provider value={appContext}>{children}</AppContext.Provider>
  );
}

// 4. Create a custom consumer hook (check Timeline.tsx for usage)
export function useAppContext() {
  const context = useContext(AppContext);

  if (context === undefined) {
    throw new Error("useAppContext must be used inside AppContextProvider!");
  }

  return context;
}
```
